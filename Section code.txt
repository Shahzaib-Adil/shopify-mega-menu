<script>
  (function () {
    var desktopQuery = window.matchMedia('(min-width: 990px)');
    var closeDelay = 120;

    function setupMegaMenu(menu) {
      if (menu.dataset.ready === 'true') return;
      menu.dataset.ready = 'true';

      var items = Array.from(menu.querySelectorAll('[data-menu-item]'));
      var burger = menu.querySelector('[data-menu-burger]');
      var closeTimer = null;

      function setItemState(item, isOpen) {
        var toggle = item.querySelector('[data-menu-toggle]');
        var video = item.querySelector('video');

        item.classList.toggle('is-open', isOpen);
        toggle.setAttribute('aria-expanded', String(isOpen));

        if (!video) return;

        if (isOpen) {
          video.play().catch(function () {});
        } else {
          video.pause();
        }
      }

      function closeAllItems(exceptItem) {
        items.forEach(function (item) {
          if (item !== exceptItem) setItemState(item, false);
        });
      }

      function closeMobileNav() {
        menu.classList.remove('is-mobile-open');
        if (burger) burger.setAttribute('aria-expanded', 'false');
      }

      items.forEach(function (item) {
        var toggle = item.querySelector('[data-menu-toggle]');

        item.addEventListener('mouseenter', function () {
          if (!desktopQuery.matches) return;
          clearTimeout(closeTimer);
          closeAllItems(item);
          setItemState(item, true);
        });

        item.addEventListener('mouseleave', function () {
          if (!desktopQuery.matches) return;
          closeTimer = setTimeout(function () {
            setItemState(item, false);
          }, closeDelay);
        });

        toggle.addEventListener('click', function () {
          var shouldOpen = !item.classList.contains('is-open');
          if (desktopQuery.matches) closeAllItems(item);
          setItemState(item, shouldOpen);
        });
      });

      if (burger) {
        burger.addEventListener('click', function () {
          var shouldOpen = !menu.classList.contains('is-mobile-open');
          menu.classList.toggle('is-mobile-open', shouldOpen);
          burger.setAttribute('aria-expanded', String(shouldOpen));
        });
      }

      document.addEventListener('click', function (event) {
        if (menu.contains(event.target)) return;
        closeAllItems();
        closeMobileNav();
      });

      menu.addEventListener('keydown', function (event) {
        if (event.key !== 'Escape') return;
        closeAllItems();
        closeMobileNav();
      });

      menu.addEventListener('focusout', function (event) {
        var nextTarget = event.relatedTarget;
        if (desktopQuery.matches && nextTarget && !menu.contains(nextTarget)) {
          closeAllItems();
        }
      });

      desktopQuery.addEventListener('change', function () {
        closeAllItems();
        closeMobileNav();
      });

      menu.querySelectorAll('video').forEach(function (video) {
        video.pause();
      });
    }

    function setupAllMenus() {
      document.querySelectorAll('[data-mega-menu]').forEach(setupMegaMenu);
    }

    setupAllMenus();
    document.addEventListener('shopify:section:load', setupAllMenus);
  })();
